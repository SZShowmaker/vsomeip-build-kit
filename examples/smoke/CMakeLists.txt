cmake_minimum_required(VERSION 3.16)
project(someip_smoke LANGUAGES CXX)
find_package(vsomeip3 3.6.4 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(someip_smoke main.cpp)
target_compile_features(someip_smoke PRIVATE cxx_std_17)
target_link_libraries(someip_smoke PRIVATE vsomeip3 Threads::Threads)
if(MSVC)
  target_compile_options(someip_smoke PRIVATE /W4 /EHsc)
  set_property(TARGET someip_smoke PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
else()
  target_compile_options(someip_smoke PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(someip_smoke PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS someip_smoke RUNTIME DESTINATION bin)
